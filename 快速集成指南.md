### 4. 在ChatInterface中添加标注模式

在 `/opt/frontend/app/[locale]/chat/internal/chatInterface.tsx` 中：

```typescript
import { ImageAnnotationView } from '@/components/common/image-annotation';

// 添加状态
const [annotationMode, setAnnotationMode] = useState(false);
const [currentImageData, setCurrentImageData] = useState<{
  imageId: number;
  imageUrl: string;
} | null>(null);

// 切换到标注模式
const enterAnnotationMode = (imageId: number, imageUrl: string) => {
  setCurrentImageData({ imageId, imageUrl });
  setAnnotationMode(true);
};

// 渲染
{annotationMode && currentImageData ? (
  <ImageAnnotationView
    imageId={currentImageData.imageId}
    imageUrl={currentImageData.imageUrl}
    onClose={() => setAnnotationMode(false)}
    chatComponent={/* 可选：右侧聊天界面 */}
  />
) : (
  // 原有的聊天界面
)}
```
